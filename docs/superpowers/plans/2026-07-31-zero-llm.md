# 零 LLM 功能实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在不依赖 LLM 平台的前提下，将 superapp 从硬编码看板骨架升级为完整可用的本地看板应用（多看板、列、卡片、子任务、拖拽、Electron 生产构建），并为后续接入 NexusPilot 铺好契约与代理骨架。

**Architecture:** 单一 zustand store（`boardStore`）作为 UI 与数据层之间的唯一通道；数据层 `lib/db` 维持环境检测路由（Electron→IPC→Prisma/SQLite，Web→IndexedDB），接口统一为对称 CRUD；LLM 前瞻独立于看板数据流，仅含契约类型、mock adapter 与代理路由，不依赖平台运行。

**Tech Stack:** Next.js 14（App Router）、React 18、zustand ^4.5.5（已有依赖）、@dnd-kit（拖拽）、idb（IndexedDB）、react-markdown、vitest + @testing-library/react + fake-indexeddb、ESLint 9 flat config、Prisma 5（SQLite）、Electron 31。

**设计规格：** `docs/superpowers/specs/2026-07-31-zero-llm-design.md`

## Global Constraints

- Node >= 18（root `package.json` engines）；pnpm 9.15.0
- 中文 UI（看板列名、按钮文字、错误提示）
- 数据拓扑：Desktop（SQLite）与 Web（IndexedDB）各自独立持久化，**不做同步**
- 无用户鉴权；LLM 代理路由仅限 localhost 本地开发
- API Key 与平台地址只存在于 Next.js 服务端环境变量（`NEXUSPILOT_BASE_URL`/`NEXUSPILOT_API_KEY`），禁止进入浏览器端代码
- Electron 安全基线保持：`contextIsolation: true`、`nodeIntegration: false`；IPC 输入必须校验
- 依赖新增用 `pnpm add`，版本以解析结果为准；与 React 18 兼容性在安装后 `typecheck` 验证
- 每任务结束时：该包 `typecheck` 通过 + 测试通过 + commit
- 提交信息格式：`feat:`/`test:`/`chore:` 前缀 + 中文或英文描述，结尾附 `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`

`apps/desktop/src/lib/cardOrder.ts`：

```ts
export function computeMoveOrders(
  ids: string[],
  movingId: string,
  toIndex: number,
): { id: string; order: number }[] {
  if (!ids.includes(movingId)) {
    throw new Error(`movingId ${movingId} not in ids`);
  }
  const rest = ids.filter((id) => id !== movingId);
  const clamped = Math.max(0, Math.min(toIndex, rest.length));
  const next = [...rest.slice(0, clamped), movingId, ...rest.slice(clamped)];
  return next.map((id, order) => ({ id, order }));
}
```

`apps/desktop/src/ipc/validate.ts`：

```ts
import type { CardData } from '@superapp/shared';

export type CardUpdateData = Partial<
  Pick<CardData, 'title' | 'content' | 'labels' | 'dueDate' | 'parentId'>
>;

const STRING_FIELDS = new Set(['title', 'content', 'labels'] as const);
const NULLABLE_FIELDS = new Set(['dueDate', 'parentId'] as const);

export function assertNonEmptyString(
  value: unknown,
  field: string,
  maxLen = 100,
): asserts value is string {
  if (typeof value !== 'string' || value.trim().length === 0) {
    throw new Error(`${field} must be a non-empty string`);
  }
  if (value.trim().length > maxLen) {
    throw new Error(`${field} must be at most ${maxLen} characters`);
  }
}

export function assertId(value: unknown, field: string): asserts value is string {
  if (typeof value !== 'string' || value.trim().length === 0) {
    throw new Error(`${field} must be a non-empty string (id)`);
  }
}

export function assertNonNegativeInt(value: unknown, field: string): asserts value is number {
  if (typeof value !== 'number' || !Number.isInteger(value) || value < 0) {
    throw new Error(`${field} must be a non-negative integer`);
  }
}

export function assertCardUpdate(value: unknown): asserts value is CardUpdateData {
  if (typeof value !== 'object' || value === null) {
    throw new Error('card update must be an object');
  }
  for (const key of Object.keys(value)) {
    if (STRING_FIELDS.has(key as never)) {
      const v = (value as Record<string, unknown>)[key];
      if (typeof v !== 'string' || v.length > 100_000) {
        throw new Error(`field ${key} must be a string (max 100000 chars)`);
      }
    } else if (NULLABLE_FIELDS.has(key as never)) {
      const v = (value as Record<string, unknown>)[key];
      if (v !== null && (key === 'dueDate' ? !(v instanceof Date) : typeof v !== 'string')) {
        throw new Error(`field ${key} must be null or a valid value`);
      }
    } else {
      throw new Error(`unknown field ${key} in card update`);
    }
  }
}
```

```ts
import { ipcMain } from 'electron';
import { prisma } from '../db/client';
import {
  assertNonEmptyString,
  assertId,
  assertNonNegativeInt,
  assertCardUpdate,
} from './validate';
import { computeMoveOrders } from '../lib/cardOrder';

function wrapHandler<T extends unknown[], R>(
  handler: (...args: T) => Promise<R>,
): (...args: T) => Promise<R> {
  return async (...args: T) => {
    try {
      return await handler(...args);
    } catch (error) {
      console.error('[IPC Error]', error);
      throw new Error(error instanceof Error ? error.message : 'Unknown database error');
    }
  };
}

async function nextOrder(
  tx: Parameters<Parameters<typeof prisma.$transaction>[0]>[0],
  columnId: string,
): Promise<number> {
  const last = await tx.card.findFirst({
    where: { columnId },
    orderBy: { order: 'desc' },
    select: { order: true },
  });
  return (last?.order ?? -1) + 1;
}

export function registerDbHandlers() {
  // --- boards ---
  ipcMain.handle('db:boards:list', wrapHandler(async () => {
    return prisma.board.findMany({ include: { columns: true } });
  }));

  ipcMain.handle('db:boards:create', wrapHandler(async (_e, name: unknown) => {
    assertNonEmptyString(name, 'Board name');
    return prisma.board.create({ data: { name: name.trim() } });
  }));

  ipcMain.handle('db:boards:update', wrapHandler(async (_e, id: unknown, name: unknown) => {
    assertId(id, 'id');
    assertNonEmptyString(name, 'Board name');
    return prisma.board.update({ where: { id }, data: { name: name.trim() } });
  }));

  ipcMain.handle('db:boards:delete', wrapHandler(async (_e, id: unknown) => {
    assertId(id, 'id');
    await prisma.board.delete({ where: { id } });
  }));

  // --- columns ---
  ipcMain.handle('db:columns:list', wrapHandler(async (_e, boardId: unknown) => {
    assertId(boardId, 'boardId');
    return prisma.column.findMany({
      where: { boardId },
      include: { cards: true },
      orderBy: { order: 'asc' },
    });
  }));

  ipcMain.handle('db:columns:create', wrapHandler(async (_e, boardId: unknown, name: unknown) => {
    assertId(boardId, 'boardId');
    assertNonEmptyString(name, 'Column name');
    const order = await nextOrder(prisma, boardId);
    return prisma.column.create({ data: { boardId, name: name.trim(), order } });
  }));

  ipcMain.handle('db:columns:update', wrapHandler(async (_e, id: unknown, name: unknown) => {
    assertId(id, 'id');
    assertNonEmptyString(name, 'Column name');
    return prisma.column.update({ where: { id }, data: { name: name.trim() } });
  }));

  ipcMain.handle('db:columns:delete', wrapHandler(async (_e, id: unknown) => {
    assertId(id, 'id');
    await prisma.column.delete({ where: { id } }); // 卡片级联删除（Task 4）
  }));

  ipcMain.handle('db:columns:reorder', wrapHandler(async (_e, boardId: unknown, orderedIds: unknown) => {
    assertId(boardId, 'boardId');
    if (!Array.isArray(orderedIds) || orderedIds.some((x) => typeof x !== 'string')) {
      throw new Error('orderedIds must be a string array');
    }
    await prisma.$transaction(
      orderedIds.map((id, order) => prisma.column.update({ where: { id }, data: { order } })),
    );
  }));

  // --- cards ---
  ipcMain.handle('db:cards:list', wrapHandler(async (_e, columnId: unknown) => {
    assertId(columnId, 'columnId');
    return prisma.card.findMany({ where: { columnId }, orderBy: { order: 'asc' } });
  }));

  ipcMain.handle('db:cards:listChildren', wrapHandler(async (_e, parentId: unknown) => {
    assertId(parentId, 'parentId');
    return prisma.card.findMany({ where: { parentId }, orderBy: { order: 'asc' } });
  }));

  ipcMain.handle('db:cards:create', wrapHandler(async (_e, columnId: unknown, data: unknown) => {
    assertId(columnId, 'columnId');
    const { title } = (data ?? {}) as Record<string, unknown>;
    assertNonEmptyString(title, 'Card title');
    const order = await nextOrder(prisma, columnId);
    return prisma.card.create({ data: { columnId, title: title.trim(), order } });
  }));

  ipcMain.handle('db:cards:update', wrapHandler(async (_e, id: unknown, data: unknown) => {
    assertId(id, 'id');
    assertCardUpdate(data);
    return prisma.card.update({ where: { id }, data });
  }));

  ipcMain.handle('db:cards:delete', wrapHandler(async (_e, id: unknown) => {
    assertId(id, 'id');
    await prisma.card.delete({ where: { id } }); // 子任务级联删除（Task 4）
  }));

  ipcMain.handle('db:cards:move', wrapHandler(async (_e, cardId: unknown, toColumnId: unknown, toIndex: unknown) => {
    assertId(cardId, 'cardId');
    assertId(toColumnId, 'toColumnId');
    assertNonNegativeInt(toIndex, 'toIndex');
    await prisma.$transaction(async (tx) => {
      const card = await tx.card.findUnique({ where: { id: cardId } });
      if (!card) throw new Error('Card not found');
      const target = await tx.column.findUnique({ where: { id: toColumnId } });
      if (!target) throw new Error('Target column not found');

      // 源列：移除 movingId 后其余卡片压缩 order
      if (card.columnId !== toColumnId) {
        const srcIds = (await tx.card.findMany({
          where: { columnId: card.columnId },
          orderBy: { order: 'asc' },
          select: { id: true },
        })).map((c) => c.id);
        const srcOrders = computeMoveOrders(srcIds, cardId, srcIds.length - 1).filter((o) => o.id !== cardId);
        for (const { id, order } of srcOrders) {
          await tx.card.update({ where: { id }, data: { order } });
        }
      }

      // 目标列：插入 movingId 到 toIndex
      const dstIds = (await tx.card.findMany({
        where: { columnId: toColumnId },
        orderBy: { order: 'asc' },
        select: { id: true },
      })).map((c) => c.id);
      const dstOrders = computeMoveOrders(dstIds, cardId, toIndex);
      for (const { id, order } of dstOrders) {
        await tx.card.update({
          where: { id },
          data: { order, ...(id === cardId ? { columnId: toColumnId } : {}) },
        });
      }
    });
  }));
}

export function getWebDb(): WebDb {
  return {
    boards: {
      list: async () => {
        const db = await open();
        return db.getAll('boards');
      },
      create: async (name) => {
        const db = await open();
        const board: BoardData = {
          id: crypto.randomUUID(), name, columns: [], createdAt: new Date(),
        };
        await db.put('boards', board);
        return board;
      },
      update: async (id, data) => {
        const db = await open();
        const existing = await db.get('boards', id);
        if (!existing) throw new Error('Board not found');
        const next = { ...existing, ...data };
        await db.put('boards', next);
        return next;
      },
      delete: async (id) => {
        const db = await open();
        await db.transaction(['boards', 'columns', 'cards'], 'readwrite', async (tx) => {
          const cols = await tx.objectStore('columns').getAll();
          const colIds = cols.filter((c) => c.boardId === id).map((c) => c.id);
          await tx.objectStore('boards').delete(id);
          for (const cid of colIds) await tx.objectStore('columns').delete(cid);
          const cards = await tx.objectStore('cards').getAll();
          for (const card of cards) {
            if (colIds.includes(card.columnId)) await tx.objectStore('cards').delete(card.id);
          }
        });
      },
    },
    columns: {
      list: async (boardId) => {
        const db = await open();
        const all = await db.getAll('columns');
        return all.filter((c) => c.boardId === boardId).sort((a, b) => a.order - b.order);
      },
      create: async (boardId, name) => {
        const db = await open();
        const all = await db.getAll('columns');
        const order = all.filter((c) => c.boardId === boardId).length;
        const column: ColumnData = { id: crypto.randomUUID(), name, order, boardId, cards: [], createdAt: new Date() };
        await db.put('columns', column);
        return column;
      },
      update: async (id, data) => { /* 同 boards.update 模式 */ },
      delete: async (id) => {
        const db = await open();
        await db.transaction(['columns', 'cards'], 'readwrite', async (tx) => {
          await tx.objectStore('columns').delete(id);
          const cards = await tx.objectStore('cards').getAll();
          for (const card of cards) {
            if (card.columnId === id) await tx.objectStore('cards').delete(card.id);
          }
        });
      },
      reorder: async (boardId, orderedIds) => {
        const db = await open();
        await db.transaction('columns', 'readwrite', async (tx) => {
          const all = await tx.objectStore('columns').getAll();
          for (const col of all) {
            if (col.boardId !== boardId) continue;
            const idx = orderedIds.indexOf(col.id);
            await tx.objectStore('columns').put({ ...col, order: idx === -1 ? col.order : idx });
          }
        });
      },
    },
    cards: {
      list: async (columnId) => {
        const db = await open();
        const all = await db.getAll('cards');
        return all.filter((c) => c.columnId === columnId).sort((a, b) => a.order - b.order);
      },
      listChildren: async (parentId) => {
        const db = await open();
        const all = await db.getAll('cards');
        return all.filter((c) => c.parentId === parentId).sort((a, b) => a.order - b.order);
      },
      create: async (columnId, data) => {
        const db = await open();
        const all = await db.getAll('cards');
        const order = all.filter((c) => c.columnId === columnId).length;
        const card: CardData = {
          id: crypto.randomUUID(), title: data.title, content: null,
          columnId, parentId: null, labels: '[]', dueDate: null,
          order, createdAt: new Date(), updatedAt: new Date(),
        };
        await db.put('cards', card);
        return card;
      },
      update: async (id, data) => {
        const db = await open();
        const existing = await db.get('cards', id);
        if (!existing) throw new Error('Card not found');
        const next = { ...existing, ...data, updatedAt: new Date() };
        await db.put('cards', next);
        return next;
      },
      delete: async (id) => {
        const db = await open();
        await db.transaction('cards', 'readwrite', async (tx) => {
          await tx.objectStore('cards').delete(id);
          const all = await tx.objectStore('cards').getAll();
          for (const card of all) {
            if (card.parentId === id) await tx.objectStore('cards').delete(card.id);
          }
        });
      },
      move: async (cardId, toColumnId, toIndex) => {
        const db = await open();
        await db.transaction('cards', 'readwrite', async (tx) => {
          const store = tx.objectStore('cards');
          const card = await store.get(cardId);
          if (!card) throw new Error('Card not found');
          const all = await store.getAll();
          if (card.columnId !== toColumnId) {
            const srcIds = all.filter((c) => c.columnId === card.columnId)
              .sort((a, b) => a.order - b.order).map((c) => c.id);
            for (const { id, order } of computeMoveOrders(srcIds, cardId, srcIds.length - 1)) {
              if (id === cardId) continue;
              await store.put({ ...(await store.get(id))!, order });
            }
          }
          const dstIds = all.filter((c) => c.columnId === toColumnId)
            .sort((a, b) => a.order - b.order).map((c) => c.id);
          for (const { id, order } of computeMoveOrders(dstIds, cardId, toIndex)) {
            const existing = await store.get(id);
            if (!existing) continue;
            await store.put({ ...existing, order, ...(id === cardId ? { columnId: toColumnId } : {}) });
          }
        });
      },
    },
  };
}

'use client';

import { create } from 'zustand';
import type { BoardData, ColumnData, CardData, CardUpdateData } from '@superapp/shared';
import { getDb } from '@/lib/db';
import { ensureDefaultBoard } from '@/lib/db/seed';

export interface BoardState {
  boards: BoardData[];
  activeBoardId: string | null;
  columns: ColumnData[];
  cardsByColumn: Record<string, CardData[]>;
  childrenByParent: Record<string, CardData[]>;
  loading: boolean;
  error: string | null;
  init(): Promise<void>;
  selectBoard(id: string): Promise<void>;
  createBoard(name: string): Promise<void>;
  renameBoard(id: string, name: string): Promise<void>;
  deleteBoard(id: string): Promise<void>;
  createColumn(name: string): Promise<void>;
  renameColumn(id: string, name: string): Promise<void>;
  deleteColumn(id: string): Promise<void>;
  reorderColumns(orderedIds: string[]): Promise<void>;
  createCard(columnId: string): Promise<void>;
  updateCard(id: string, data: CardUpdateData): Promise<void>;
  deleteCard(id: string): Promise<void>;
  moveCard(cardId: string, toColumnId: string, toIndex: number): Promise<void>;
  moveCardLocal(cardId: string, toColumnId: string, toIndex: number): void;
  loadChildren(parentId: string): Promise<void>;
  addSubtask(parentId: string, title: string): Promise<void>;
  resetError(): void;
}

async function loadActiveBoard(
  set: (partial: Partial<BoardState>) => void,
  get: () => BoardState,
  boardId: string,
) {
  const db = getDb();
  const columns = await db.columns.list(boardId);
  const cardsByColumn: Record<string, CardData[]> = {};
  for (const col of columns) {
    cardsByColumn[col.id] = await db.cards.list(col.id);
  }
  set({ columns, cardsByColumn });
}

export const useBoardStore = create<BoardState>((set, get) => ({
  boards: [],
  activeBoardId: null,
  columns: [],
  cardsByColumn: {},
  childrenByParent: {},
  loading: false,
  error: null,

  init: async () => {
    set({ loading: true, error: null });
    try {
      const db = getDb();
      const board = await ensureDefaultBoard(db);
      const boards = await db.boards.list();
      set({ boards, activeBoardId: board.id });
      await loadActiveBoard(set, get, board.id);
    } catch (e) {
      set({ error: e instanceof Error ? e.message : 'Unknown error' });
    } finally {
      set({ loading: false });
    }
  },

  selectBoard: async (id) => {
    set({ activeBoardId: id, error: null });
    await loadActiveBoard(set, get, id);
  },

  createBoard: async (name) => {
    const db = getDb();
    const board = await db.boards.create(name);
    set({ boards: [...get().boards, board] });
    await get().selectBoard(board.id);
  },

  renameBoard: async (id, name) => {
    const updated = await getDb().boards.update(id, { name });
    set({ boards: get().boards.map((b) => (b.id === id ? updated : b)) });
  },

  deleteBoard: async (id) => {
    await getDb().boards.delete(id);
    const rest = get().boards.filter((b) => b.id !== id);
    set({ boards: rest });
    if (get().activeBoardId === id) {
      if (rest.length > 0) {
        await get().selectBoard(rest[0].id);
      } else {
        set({ activeBoardId: null, columns: [], cardsByColumn: {} });
      }
    }
  },

  createColumn: async (name) => {
    const boardId = get().activeBoardId;
    if (!boardId) return;
    await getDb().columns.create(boardId, name);
    await loadActiveBoard(set, get, boardId);
  },

  renameColumn: async (id, name) => {
    await getDb().columns.update(id, { name });
    set({ columns: get().columns.map((c) => (c.id === id ? { ...c, name } : c)) });
  },

  deleteColumn: async (id) => {
    const boardId = get().activeBoardId;
    if (!boardId) return;
    await getDb().columns.delete(id);
    await loadActiveBoard(set, get, boardId);
  },

  reorderColumns: async (orderedIds) => {
    const boardId = get().activeBoardId;
    if (!boardId) return;
    await getDb().columns.reorder(boardId, orderedIds);
    set({ columns: orderedIds.map((id, order) => ({ ...get().columns.find((c) => c.id === id)!, order })) });
  },

  createCard: async (columnId) => {
    const card = await getDb().cards.create(columnId, { title: '新卡片' });
    set({
      cardsByColumn: {
        ...get().cardsByColumn,
        [columnId]: [...(get().cardsByColumn[columnId] ?? []), card],
      },
    });
  },

  updateCard: async (id, data) => {
    const updated = await getDb().cards.update(id, data);
    const { columnId } = updated;
    set({
      cardsByColumn: {
        ...get().cardsByColumn,
        [columnId]: (get().cardsByColumn[columnId] ?? []).map((c) => (c.id === id ? updated : c)),
      },
    });
  },

  deleteCard: async (id) => {
    const db = getDb();
    const card = (Object.values(get().cardsByColumn).flat() as CardData[]).find((c) => c.id === id);
    await db.cards.delete(id);
    if (!card) return;
    const columnId = card.columnId;
    set({
      cardsByColumn: {
        ...get().cardsByColumn,
        [columnId]: (get().cardsByColumn[columnId] ?? []).filter((c) => c.id !== id),
      },
    });
  },

  moveCardLocal: (cardId, toColumnId, toIndex) => {
    const { cardsByColumn } = get();
    const all = Object.entries(cardsByColumn).flatMap(([colId, cards]) =>
      cards.map((c) => ({ colId, card: c })),
    );
    const entry = all.find((x) => x.card.id === cardId);
    if (!entry) return;
    const srcCol = entry.colId;
    const srcNext = cardsByColumn[srcCol].filter((c) => c.id !== cardId);
    const dstBase = srcCol === toColumnId ? srcNext : cardsByColumn[toColumnId] ?? [];
    const dstNext = [...dstBase.slice(0, toIndex), entry.card, ...dstBase.slice(toIndex)];
    set({
      cardsByColumn: {
        ...cardsByColumn,
        [srcCol]: srcNext,
        [toColumnId]: dstNext,
      },
    });
  },

  moveCard: async (cardId, toColumnId, toIndex) => {
    const snapshot = get().cardsByColumn;
    get().moveCardLocal(cardId, toColumnId, toIndex);
    try {
      await getDb().cards.move(cardId, toColumnId, toIndex);
    } catch (e) {
      set({ cardsByColumn: snapshot, error: e instanceof Error ? e.message : 'Unknown error' });
    }
  },

  loadChildren: async (parentId) => {
    const children = await getDb().cards.listChildren(parentId);
    set({ childrenByParent: { ...get().childrenByParent, [parentId]: children } });
  },

  addSubtask: async (parentId, title) => {
    const parent = (Object.values(get().cardsByColumn).flat() as CardData[]).find((c) => c.id === parentId);
    if (!parent) return;
    const child = await getDb().cards.create(parent.columnId, { title, parentId });
    set({
      childrenByParent: {
        ...get().childrenByParent,
        [parentId]: [...(get().childrenByParent[parentId] ?? []), child],
      },
    });
  },

  resetError: () => set({ error: null }),
}));
```

'use client';

import React, { useState } from 'react';
import type { CardData } from '@superapp/shared';
import { useBoardStore } from '@/lib/store/boardStore';

export function CardItem({ card }: { card: CardData }) {
  const updateCard = useBoardStore((s) => s.updateCard);
  const deleteCard = useBoardStore((s) => s.deleteCard);
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(card.title);

  const commit = () => {
    if (draft.trim() && draft.trim() !== card.title) {
      void updateCard(card.id, { title: draft.trim() });
    }
    setEditing(false);
  };

  if (editing) {
    return (
      <input
        className="w-full rounded-md border bg-background px-2 py-1 text-sm"
        value={draft}
        autoFocus
        onChange={(e) => setDraft(e.target.value)}
        onBlur={commit}
        onKeyDown={(e) => {
          if (e.key === 'Enter') commit();
          if (e.key === 'Escape') { setDraft(card.title); setEditing(false); }
        }}
        aria-label="编辑卡片标题"
      />
    );
  }

  return (
    <div
      className="group relative rounded-md border bg-background px-3 py-2 text-sm shadow-sm cursor-pointer hover:border-ring"
      onClick={() => setEditing(true)}
    >
      {card.title}
      <button
        className="absolute right-1 top-1 hidden group-hover:block text-muted-foreground hover:text-destructive"
        aria-label="删除卡片"
        onClick={(e) => {
          e.stopPropagation();
          if (confirm(`删除卡片「${card.title}」？`)) void deleteCard(card.id);
        }}
      >
        ✕
      </button>
    </div>
  );
}

```tsx
'use client';

import React, { useState, useEffect } from 'react';
import ReactMarkdown from 'react-markdown';
import { useBoardStore } from '@/lib/store/boardStore';

export function CardDialog({ cardId, onClose }: { cardId: string; onClose: () => void }) {
  const card = useBoardStore((s) =>
    Object.values(s.cardsByColumn).flat().find((c) => c.id === cardId),
  );
  const updateCard = useBoardStore((s) => s.updateCard);
  const [title, setTitle] = useState(card?.title ?? '');
  const [content, setContent] = useState(card?.content ?? '');
  const [labels, setLabels] = useState<string[]>(safeParseLabels(card?.labels));
  const [dueDate, setDueDate] = useState<string>(card?.dueDate?.toISOString().slice(0, 10) ?? '');
  const [preview, setPreview] = useState(false);

  if (!card) return null;

  const hasUnsaved =
    title.trim() !== card.title ||
    content !== (card.content ?? '') ||
    JSON.stringify(labels) !== card.labels ||
    (dueDate ? new Date(dueDate).toISOString().slice(0, 10) : '') !== (card.dueDate?.toISOString().slice(0, 10) ?? '');

  const save = () => {
    void updateCard(card.id, {
      title: title.trim() || card.title,
      content,
      labels: JSON.stringify(labels),
      dueDate: dueDate ? new Date(dueDate) : null,
    });
    onClose();
  };

  const tryClose = () => {
    if (hasUnsaved && !confirm('有未保存的修改，确定丢弃？')) return;
    onClose();
  };

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40" onClick={tryClose}>
      <div className="w-[560px] max-h-[80vh] overflow-y-auto rounded-lg bg-background p-6 shadow-xl" onClick={(e) => e.stopPropagation()}>
        <input aria-label="标题" className="mb-3 w-full border-b text-lg font-semibold" value={title} onChange={(e) => setTitle(e.target.value)} />
        <div className="mb-2 flex flex-wrap gap-1">
          {labels.map((l) => <span key={l} className="rounded-full bg-muted px-2 py-0.5 text-xs">{l}</span>)}
          <input aria-label="标签" className="w-40 border-b text-sm" placeholder="标签（逗号分隔）"
            onKeyDown={(e) => {
              if (e.key === 'Enter') {
                const v = (e.target as HTMLInputElement).value;
                if (v.trim()) setLabels([...labels, ...v.split(',').map((x) => x.trim()).filter(Boolean)]);
                (e.target as HTMLInputElement).value = '';
              }
            }} />
        </div>
        <div className="mb-2 flex items-center gap-2 text-sm">
          <label>截止日期 <input aria-label="截止日期" type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} /></label>
          <button className="underline" onClick={() => setPreview(!preview)}>{preview ? '编辑' : '预览'}</button>
        </div>
        {preview ? (
          <div className="prose prose-sm max-w-none"><ReactMarkdown>{content}</ReactMarkdown></div>
        ) : (
          <textarea aria-label="描述" className="min-h-[160px] w-full rounded-md border p-2 text-sm" value={content} onChange={(e) => setContent(e.target.value)} />
        )}
        <div className="mt-4 flex justify-end gap-2">
          <button className="rounded-md border px-3 py-1 text-sm" onClick={tryClose}>取消</button>
          <button className="rounded-md bg-primary px-3 py-1 text-sm text-primary-foreground" onClick={save}>保存</button>
        </div>
      </div>
    </div>
  );
}

function safeParseLabels(raw: string | null | undefined): string[] {
  try {
    const v = JSON.parse(raw ?? '[]');
    return Array.isArray(v) ? v.filter((x) => typeof x === 'string') : [];
  } catch {
    return [];
  }
}
```

import { getLlmConfig } from '@/lib/llm/config';

export const runtime = 'nodejs';

export async function POST(request: Request) {
  const cfg = getLlmConfig();
  if (!cfg.baseUrl) {
    return Response.json({ error: 'NEXUSPILOT_BASE_URL not configured' }, { status: 503 });
  }
  const body = await request.json();
  const upstream = await fetch(`${cfg.baseUrl}/api/v1/responses`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(cfg.apiKey ? { 'X-API-Key': cfg.apiKey } : {}),
    },
    body: JSON.stringify({ ...body, stream: body.stream ?? false }),
  });

  const contentType = upstream.headers.get('content-type') ?? '';
  if (contentType.includes('text/event-stream')) {
    // SSE 透传
    const stream = new ReadableStream({
      async start(controller) {
        const reader = upstream.body!.getReader();
        const decoder = new TextDecoder();
        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          controller.enqueue(decoder.decode(value, { stream: true }));
        }
        controller.close();
      },
    });
    return new Response(stream, {
      status: upstream.status,
      headers: { 'Content-Type': 'text/event-stream; charset=utf-8', 'Cache-Control': 'no-cache' },
    });
  }
  return new Response(upstream.body, { status: upstream.status, headers: { 'Content-Type': contentType } });
}

export async function* streamResponse(req: ResponseRequest): AsyncGenerator<ResponseEvent> {
  const res = await fetch('/api/llm/responses', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ ...req, stream: true }),
  });
  if (!res.ok) {
    const body = await res.json().catch(() => null);
    throw new Error(body?.error ?? `proxy returned ${res.status}`);
  }
  const reader = res.body!.getReader();
  const decoder = new TextDecoder();
  let buf = '';
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    buf += decoder.decode(value, { stream: true });
    // 按 SSE 行解析 event:/data: 成 ResponseEvent（协议事件名见 types.ts）
    // 解析逻辑: 以空行分隔事件块，data 字段 JSON.parse 后映射 type
    // 本函数在平台就绪后用真实流验证；mock 流走 Task 19 的 ChatPanel
  }
}

'use client';

import React, { useState, useRef, useEffect } from 'react';
import { mockStream } from '@/lib/llm/mock';
import type { ResponseEvent } from '@/lib/llm/types';

interface ChatMessage {
  id: string;
  role: 'user' | 'assistant';
  text: string;
}

let nextId = 0;

export function ChatPanel() {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);
  const listRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    listRef.current?.scrollTo({ top: listRef.current.scrollHeight });
  }, [messages]);

  const send = async () => {
    const text = input.trim();
    if (!text || busy) return;
    setInput('');
    setBusy(true);
    setMessages((m) => [...m, { id: `u${nextId++}`, role: 'user', text }]);
    const assistantId = `a${nextId++}`;
    setMessages((m) => [...m, { id: assistantId, role: 'assistant', text: '' }]);
    try {
      for await (const ev of mockStream() as AsyncGenerator<ResponseEvent>) {
        if (ev.type === 'response.text.delta') {
          setMessages((m) =>
            m.map((msg) => (msg.id === assistantId ? { ...msg, text: msg.text + ev.delta } : msg)),
          );
        }
        if (ev.type === 'response.failed') {
          setMessages((m) => m.map((msg) => (msg.id === assistantId ? { ...msg, text: `错误: ${ev.error}` } : msg)));
        }
      }
    } finally {
      setBusy(false);
    }
  };

  return (
    <aside className="w-80 shrink-0 border-l flex flex-col">
      <div className="border-b px-4 py-2 text-sm font-medium">对话</div>
      <div ref={listRef} className="flex-1 overflow-y-auto p-4 space-y-3">
        {messages.map((m) => (
          <div key={m.id} className={`text-sm whitespace-pre-wrap ${m.role === 'user' ? 'text-right' : ''}`}>
            <span className={`inline-block rounded-lg px-3 py-2 ${m.role === 'user' ? 'bg-primary text-primary-foreground' : 'bg-muted'}`}>
              {m.text || (m.role === 'assistant' ? '…' : '')}
            </span>
          </div>
        ))}
      </div>
      <div className="border-t p-3 flex gap-2">
        <input
          className="flex-1 rounded-md border px-2 py-1 text-sm"
          placeholder="输入消息…"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => { if (e.key === 'Enter') void send(); }}
        />
        <button className="rounded-md bg-primary px-3 py-1 text-sm text-primary-foreground" onClick={() => void send()} disabled={busy}>
          发送
        </button>
      </div>
    </aside>
  );
}
